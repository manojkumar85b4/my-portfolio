SELECT c.course_name,
       COUNT(e.student_id) AS enrolled
FROM courses c
LEFT JOIN enrollments e
  ON e.course_id = c.course_id
GROUP BY c.course_name
HAVING COUNT(e.student_id) > 0
ORDER BY enrolled DESC
LIMIT 5;
